# Proposed GitHub Actions jobs for the process suite (crates/node/tests/fault/suite.py),
# replacing merobox's cloud-v2 backend. NOT wired in: the integrator moves each block
# into the workflow named above it. Action pins are copied from e2e.yml/release.yml.
#
# What each replaces:
#   e2e.yml  `examples` (merobox examples)          -> `process-suite`
#   e2e.yml  "Soak latency and throughput" steps     -> the same step, on suite.py's report
#   e2e.yml  `flamegraph`                            -> `flamegraph` (scripts/soak-flamegraph.sh, rewritten)
#   e2e.yml  `release-*` (Latest release image)      -> `release-process-suite` (after `release-image`)
#   release.yml `image-e2e` "merobox's fault examples" + soak -> "Process suite (smoke)" steps
#
# The suite needs release merod and meroctl, kv_store.wasm and, for `packages`, two
# signed releases of kv-store. Release builds: the suite's timing budgets (frozen
# leader: 10 s) and the soak's latencies are those of an optimized node.

# ---------------------------------------------------------------- e2e.yml

  # Process-level tests on real `merod run --dev` processes, one per node: a crash,
  # a frozen node and a restart; a frozen leader under a 10 s budget; organizations
  # and keys; signed packages from a registry; a rolling upgrade of every node under
  # load; the kv-store soak (latency and throughput in the summary). A PR runs each
  # once with a 30 s soak; a nightly runs `--nightly` (three iterations, a 900 s
  # soak, rooms_fault.py's randomized nemesis for 50 iterations).
  process-suite:
    name: process suite (merod processes)
    runs-on: big-machine
    timeout-minutes: ${{ (github.event_name == 'schedule' || inputs.nightly) && 150 || 45 }}
    steps:
      - name: Checkout code
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ github.event.pull_request.head.sha }}
          persist-credentials: false

      - name: Setup Rust CI
        uses: ./.github/actions/setup-rust-ci
        with:
          # wasm32-wasip2: the Rooms component the nightly's rooms-fault builds.
          targets: wasm32-unknown-unknown,wasm32-wasip2
          shared-key: release
          save-if: false

      - name: Setup cargo mero
        uses: ./.github/actions/setup-cargo-mero

      - uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97 # v7.0.0
        with:
          python-version: "3.12"

      - name: Build merod, meroctl and the fixtures
        env:
          NIGHTLY: ${{ (github.event_name == 'schedule' || inputs.nightly) && 'true' || 'false' }}
        run: |
          set -euo pipefail
          cargo build --release -p merod -p meroctl
          cargo build -p kv-store --target wasm32-unknown-unknown --profile app-release
          for v in 1.0.0 1.1.0; do
            (cd apps/kv-store && cargo mero bundle --dev --no-icon --app-version "$v")
          done
          if [ "$NIGHTLY" = true ]; then
            # Builds the Rooms component into target/app-components (tests/common/mod.rs).
            cargo test --release -p calimero-node --test rooms
          fi

      - name: Process suite
        env:
          SOAK_SECONDS: ${{ inputs.soak_seconds }}
          NIGHTLY: ${{ (github.event_name == 'schedule' || inputs.nightly) && '--nightly' || '' }}
        run: |
          python3 -I crates/node/tests/fault/suite.py $NIGHTLY \
            --soak-report target/process-suite/soak.json --json target/process-suite/results.json

      - name: Soak latency and throughput
        if: ${{ !cancelled() }}
        run: bash scripts/soak-summary.sh target/process-suite/soak.json "Soak, kv-store (this tree's merod, release build)"

      - name: Upload the results
        if: ${{ !cancelled() }}
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: process-suite
          # Failed tests keep their work directory: node logs (nodeN.log) and data.
          path: |
            target/process-suite
            target/fault-suite
          if-no-files-found: ignore
          retention-days: 14

  # Nightly: where one node's CPU goes during the kv-store soak.
  flamegraph:
    name: merod flamegraph (kv-store soak)
    if: github.event_name == 'schedule' || (github.event_name == 'workflow_dispatch' && (inputs.nightly || inputs.flamegraph))
    runs-on: big-machine
    timeout-minutes: 75
    steps:
      - name: Checkout code
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false

      - name: Setup Rust CI
        uses: ./.github/actions/setup-rust-ci
        with:
          targets: wasm32-unknown-unknown
          shared-key: release
          save-if: false

      - name: Build merod and meroctl for profiling, and kv-store
        env:
          RUSTFLAGS: -C force-frame-pointers=yes
        run: |
          set -euo pipefail
          cargo build -p merod -p meroctl --profile profiling
          RUSTFLAGS= cargo build -p kv-store --target wasm32-unknown-unknown --profile app-release

      - uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97 # v7.0.0
        with:
          python-version: "3.12"

      - name: Install perf
        run: |
          set -euo pipefail
          sudo apt-get update -q
          sudo apt-get install -y -q --no-install-recommends linux-tools-common linux-tools-generic "linux-tools-$(uname -r)" \
            || sudo apt-get install -y -q --no-install-recommends linux-tools-common linux-tools-generic
          perf="/usr/lib/linux-tools/$(uname -r)/perf"
          [ -x "$perf" ] || perf=$(find /usr/lib/linux-tools -name perf -type f | sort -V | tail -n1)
          "$perf" --version
          echo "PERF=$perf" >> "$GITHUB_ENV"
          sudo sysctl -w kernel.perf_event_paranoid=-1 kernel.kptr_restrict=0

      - name: Restore inferno
        id: inferno
        uses: actions/cache@55cc8345863c7cc4c66a329aec7e433d2d1c52a9 # v6.1.0
        with:
          path: ~/.inferno
          key: inferno-${{ env.INFERNO_VERSION }}-${{ runner.os }}-${{ runner.arch }}

      - name: Install inferno
        if: steps.inferno.outputs.cache-hit != 'true'
        env:
          CARGO_TARGET_DIR: ${{ runner.temp }}/inferno-build
        run: cargo install inferno --version "$INFERNO_VERSION" --locked --root ~/.inferno

      - name: Profile node 1 through the kv-store soak
        env:
          MEROD_BINARY: ${{ github.workspace }}/target/profiling/merod
          MEROCTL_BINARY: ${{ github.workspace }}/target/profiling/meroctl
          SOAK_SECONDS: "300"
        run: |
          PATH="$HOME/.inferno/bin:$PATH"
          bash scripts/soak-flamegraph.sh target/flamegraph

      - name: Summary
        if: ${{ !cancelled() }}
        run: |
          bash scripts/soak-summary.sh target/flamegraph/soak.json "Soak while profiled (profiling build, node 1 under perf)"
          if [ -s target/flamegraph/merod-flamegraph.svg ]; then
            echo "Flamegraph: artifact \`merod-flamegraph\` (merod-flamegraph.svg; merod.folded for inferno-diff-folded)." >> "$GITHUB_STEP_SUMMARY"
          fi

      - name: Upload the flamegraph
        if: ${{ !cancelled() }}
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: merod-flamegraph
          path: target/flamegraph
          if-no-files-found: ignore
          retention-days: 30

  # Nightly: the latest release image's merod (job `release-image`, unchanged, but its
  # `testable` check looks for crates/node/tests/fault/suite.py at the image's
  # revision instead of scripts/run-workflows.py). The suite, meroctl and the
  # fixtures are that commit's, so a test added after the release cannot fail it.
  release-process-suite:
    name: process suite on release ${{ needs.release-image.outputs.version }}
    needs: release-image
    if: needs.release-image.outputs.testable == 'true'
    runs-on: big-machine
    timeout-minutes: 90
    steps:
      - name: Checkout the release's commit
        uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          ref: ${{ needs.release-image.outputs.revision }}
          persist-credentials: false

      - name: Setup Rust CI
        uses: ./.github/actions/setup-rust-ci
        with:
          targets: wasm32-unknown-unknown
          shared-key: release
          save-if: false

      - name: Setup cargo mero
        uses: ./.github/actions/setup-cargo-mero

      - uses: actions/setup-python@5fda3b95a4ea91299a34e894583c3862153e4b97 # v7.0.0
        with:
          python-version: "3.12"

      - name: Build meroctl and the fixtures (the release's commit)
        run: |
          set -euo pipefail
          cargo build --release -p meroctl
          cargo build -p kv-store --target wasm32-unknown-unknown --profile app-release
          for v in 1.0.0 1.1.0; do
            (cd apps/kv-store && cargo mero bundle --dev --no-icon --app-version "$v")
          done

      - name: Download the release's merod
        uses: actions/download-artifact@9000827ccba6bdab643e8b6fd33ac0654aef8333 # v8.0.2
        with:
          name: release-merod
          path: target/release-image

      - name: Process suite (every test but the upgrade, a 900 s soak)
        env:
          VERSION: ${{ needs.release-image.outputs.version }}
        run: |
          chmod +x target/release-image/merod
          python3 -I crates/node/tests/fault/suite.py \
            --only kv-failover,frozen-leader,orgs-and-keys,packages,kv-soak --soak-seconds 900 \
            --merod target/release-image/merod --meroctl "${CARGO_TARGET_DIR:-target}/release/meroctl" \
            --soak-report target/release-image/soak.json --json target/release-image/results.json

      - name: Soak latency and throughput
        if: ${{ !cancelled() }}
        env:
          VERSION: ${{ needs.release-image.outputs.version }}
        run: bash scripts/soak-summary.sh target/release-image/soak.json "Soak, kv-store (release $VERSION's merod)"

      - name: Upload the results
        if: ${{ !cancelled() }}
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: release-process-suite
          path: |
            target/release-image/results.json
            target/release-image/soak.json
            target/fault-suite
          if-no-files-found: ignore
          retention-days: 30

# ---------------------------------------------------------------- release.yml, job image-e2e
#
# Replace the merobox checkout/install and the steps "Smoke workflows" (its merobox
# part), "merobox's fault examples" and "Soak latency and throughput" with these. The
# image's merod is in MEROD_BINARY (step "Take merod out of the image", unchanged);
# meroctl and kv-store come from this tree. `--smoke` runs kv-failover, frozen-leader,
# orgs-and-keys, packages and a 30 s kv-soak (`suite.py --list` marks them); no
# rolling upgrade: there is one image. `merod local` is not used: it runs every node
# in one process, so it cannot crash or freeze one, and the image's entry point is
# `merod run`.

      - name: Build meroctl and the suite's fixtures
        run: |
          set -euo pipefail
          cargo build --release -p meroctl
          cargo build -p kv-store --target wasm32-unknown-unknown --profile app-release
          for v in 1.0.0 1.1.0; do
            (cd apps/kv-store && cargo mero bundle --dev --no-icon --app-version "$v")
          done

      - name: Process suite (smoke) on the image's merod
        run: |
          python3 -I crates/node/tests/fault/suite.py --smoke \
            --merod "$MEROD_BINARY" --meroctl "${CARGO_TARGET_DIR:-target}/release/meroctl" \
            --soak-report target/image/soak.json --json target/image/suite.json

      - name: Soak latency and throughput
        if: ${{ !cancelled() }}
        run: bash scripts/soak-summary.sh target/image/soak.json "Soak on this run's merod image (process suite smoke, 30s)"

      # (Upload the results: add target/image/suite.json and target/fault-suite to its paths.)
